Contexte général
Ce projet est une mini station météo intelligente destinée aux agriculteurs, développé dans le cadre d'un projet IoT (Internet des Objets) académique ou professionnel, probablement au Sénégal vu les régions mentionnées (Thiès, Kaolack, Saint-Louis, Dakar, Fatick, Diourbel...).

La Station Météo Agricole est une plateforme IoT complète permettant la surveillance en temps réel des conditions météorologiques et pédologiques de parcelles agricoles au Sénégal. Chaque station physique est constituée d'un microcontrôleur ESP32 équipé de capteurs (température, humidité de l'air, humidité du sol, vitesse du vent) qui envoient leurs données vers Firebase Realtime Database via WiFi. Une interface web Streamlit, avec serveur backend nécessaire, permet à chaque agriculteur de consulter exclusivement les données de sa propre station, tandis qu'un administrateur dispose d'un tableau de bord global.
Chaque capteur est soumis à deux niveaux de traitement avant envoi : un filtre de seuil qui rejette toute valeur hors de la plage physique réaliste (parasites électriques, erreurs de lecture), puis un filtre de moyenne glissante sur les 5 dernières mesures valides pour lisser les variations brusques.
Chaque ESP32 déployé sur le terrain est identifié par un STATION_ID unique défini à la compilation. Ce champ détermine automatiquement les chemins d'écriture dans Firebase, permettant à plusieurs stations d'écrire simultanément sans conflit.
Reconnexion WiFi automatique
La boucle loop() vérifie l'état WiFi à chaque itération. En cas de perte de signal (déconnexion réseau, redémarrage du routeur), l'ESP32 appelle WiFi.reconnect() automatiquement sans nécessiter de redémarrage physique.

String obtenirTimestamp() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return "inconnu_" + String(millis());
  }
  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &timeinfo);
  return String(buf);
}


// ── INTERRUPTION ANÉMOMÈTRE ──────────────────────────────────

void IRAM_ATTR onImpulsion() {
  compteur++;
}


// ── FILTRE SEUIL + MOYENNE GLISSANTE ─────────────────────────

float filtrerValeur(float valeur, float valMin, float valMax,
                    float* hist, int &idx, int &nbV, const char* nom) {

  // Hors plage → parasite rejeté, on retourne la dernière moyenne
  if (isnan(valeur) || valeur < valMin || valeur > valMax) {
    Serial.printf("⚠ Parasite [%s]: %.2f (plage %.0f-%.0f)\n",
                  nom, valeur, valMin, valMax);
    if (nbV == 0) return (valMin + valMax) / 2.0f;
    float s = 0;
    int n = min(nbV, NB_MESURES);
    for (int i = 0; i < n; i++) s += hist[i];
    return s / n;
  }

  // Stocker dans le buffer circulaire
  hist[idx] = valeur;
  idx = (idx + 1) % NB_MESURES;
  if (nbV < NB_MESURES) nbV++;

  // Calculer la moyenne
  float s = 0;
  int n = min(nbV, NB_MESURES);
  for (int i = 0; i < n; i++) s += hist[i];
  float moy = s / n;

  Serial.printf("   [%s] brut:%.2f → lissé:%.2f\n", nom, valeur, moy);
  return moy;
}


// ── LECTURE CAPTEURS ──────────────────────────────────────────

float lireTemperature() {
  float brut = dht.readTemperature();
  if (isnan(brut)) {
    Serial.println("✗ DHT11 temperature erreur");
    return (nbTemp > 0) ? histTemp[(idxTemp - 1 + NB_MESURES) % NB_MESURES] : 25.0f;
  }
  return filtrerValeur(brut, TEMP_MIN, TEMP_MAX, histTemp, idxTemp, nbTemp, "Temp");
}

float lireHumiditeAir() {
  float brut = dht.readHumidity();
  if (isnan(brut)) {
    Serial.println("✗ DHT11 humidité air erreur");
    return (nbHumAir > 0) ? histHumAir[(idxHumAir - 1 + NB_MESURES) % NB_MESURES] : 50.0f;
  }
  return filtrerValeur(brut, HUM_AIR_MIN, HUM_AIR_MAX, histHumAir, idxHumAir, nbHumAir, "HumAir");
}

float lireHumiditeSol() {
  int   raw  = analogRead(SOL_PIN);
  // 4095 = sec (0%), 1500 = saturé (100%)
  float brut = (float)constrain(map(raw, 4095, 1500, 0, 100), 0, 100);
  return filtrerValeur(brut, HUM_SOL_MIN, HUM_SOL_MAX, histHumSol, idxHumSol, nbHumSol, "HumSol");
}

float lireVitesseVent() {
  unsigned long maintenant = millis();
  unsigned long duree      = maintenant - dernierCalcul;

  noInterrupts();
  unsigned long imp = compteur;
  compteur = 0;
  interrupts();
  dernierCalcul = maintenant;

  float hz  = (duree > 0) ? (float)imp / (duree / 1000.0f) : 0.0f;
  float kmh = hz * FACTEUR_VENT * 3.6f;

  return filtrerValeur(kmh, 0.0f, VENT_MAX, histVent, idxVent, nbVent, "Vent");
}

// Envoie un PUT sur un champ unique (écrase la valeur précédente)
// URL : https://...firebaseio.com/stations/ST002/mesures/temperature.json
bool firebasePUT(String chemin, String jsonBody) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("✗ WiFi non connecté — PUT annulé");
    return false;
  }

  HTTPClient http;
  String url = String(FIREBASE_URL) + "/" + chemin + ".json";
  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  int code = http.PUT(jsonBody);
  bool ok  = (code == 200);

  if (ok) {
    Serial.println("✓ PUT " + chemin + " → HTTP " + String(code));
  } else {
    Serial.println("✗ PUT " + chemin + " → HTTP " + String(code));
    String resp = http.getString();
    Serial.println("  Réponse : " + resp.substring(0, 100));
  }
  http.end();
  return ok;
}

// Envoie un POST (crée une clé unique push — historique grandit)
// URL : https://...firebaseio.com/stations/ST002/historique.json
bool firebasePOST(String chemin, String jsonBody) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("✗ WiFi non connecté — POST annulé");
    return false;
  }

  HTTPClient http;
  String url = String(FIREBASE_URL) + "/" + chemin + ".json";
  http.begin(url);
  http.addHeader("Content-Type", "application/json");

  int code = http.POST(jsonBody);
  bool ok  = (code == 200);

  if (ok) {
    Serial.println("✓ POST " + chemin + " → HTTP " + String(code));
  } else {
    Serial.println("✗ POST " + chemin + " → HTTP " + String(code));
    String resp = http.getString();
    Serial.println("  Réponse : " + resp.substring(0, 100));
  }
  http.end();
  return ok;
}


// Envoie les 4 mesures actuelles dans stations/ST002/mesures/
void envoyerMesures(float temp, float humAir, float humSol, float vent) {

  // → 1 seul appel HTTP au lieu de 4
  String json = "{";
  json += "\"temperature\":"  + String(temp,   2) + ",";

// GPS SIM7600 — parser NMEA $GPRMC/$GNRMC
// ============================================================

#if GPS_ACTIF
#include <HardwareSerial.h>
HardwareSerial gpsSerial(1);

bool lireGPS(float &latitude, float &longitude, float &altitude) {
  String ligne = "";
  unsigned long debut = millis();

  while (millis() - debut < 2000) {
    if (gpsSerial.available()) {
      char c = gpsSerial.read();
      if (c == '\n') {
        if (ligne.startsWith("$GPRMC") || ligne.startsWith("$GNRMC")) {
          // Parser les champs séparés par virgules
          int v[12];
          int nb = 0;
          for (int i = 0; i < (int)ligne.length() && nb < 12; i++) {
            if (ligne[i] == ',') v[nb++] = i;
          }
          if (nb >= 6 && ligne.substring(v[1]+1, v[2]) == "A") {
            // Latitude DDMM.MMMM → degrés décimaux
            String latS = ligne.substring(v[2]+1, v[3]);
            String latD = ligne.substring(v[3]+1, v[4]);
            String lonS = ligne.substring(v[4]+1, v[5]);
            String lonD = ligne.substring(v[5]+1, v[6]);

            latitude = latS.substring(0,2).toFloat()
                     + latS.substring(2).toFloat() / 60.0f;
            if (latD == "S") latitude = -latitude;

            longitude = lonS.substring(0,3).toFloat()
                      + lonS.substring(3).toFloat() / 60.0f;
            if (lonD == "W") longitude = -longitude;

            altitude = 0;
            return true;
          }
        }
        ligne = "";
      } else if (c != '\r') {
        ligne += c;
      }
    }
  }
  return false;
}
#endif


// ============================================================
// SETUP
// ============================================================

void setup() {
  Serial.begin(115200);
  delay(500);

  // ── Construire les chemins Firebase ─────────────────────────
  // On le fait ici (runtime) pour éviter les problèmes de
  // concaténation de #define strings sur ESP32
  CHEMIN_MESURES    = "stations/" + String(STATION_ID) + "/mesures";
  CHEMIN_HISTORIQUE = "stations/" + String(STATION_ID) + "/historique";
  CHEMIN_GPS        = "stations/" + String(STATION_ID) + "/gps";

  dht.begin();
  pinMode(ANEMOMETRE_PIN, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(ANEMOMETRE_PIN), onImpulsion, FALLING);

#if GPS_ACTIF
  gpsSerial.begin(9600, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  Serial.println("GPS SIM7600 init OK");
#endif

  // ── WiFi ─────────────────────────────────────────────────────
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("WiFi [" + String(WIFI_SSID) + "]");
  int tentatives = 0;
  while (WiFi.status() != WL_CONNECTED && tentatives < 30) {
    delay(500);
    Serial.print(".");
    tentatives++;
  }
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n✓ WiFi OK : " + WiFi.localIP().toString());
  } else {
    Serial.println("\n✗ WiFi ECHEC — redémarrage dans 5s");
    delay(5000);
    ESP.restart();
  }

  // ── NTP (heure réseau) ───────────────────────────────────────
  configTime(0, 0, "pool.ntp.org", "time.nist.gov");
  setenv("TZ", "GMT0", 1);
  tzset();

  Serial.print("NTP sync");
  struct tm timeinfo;
  int ntpTentatives = 0;
  while (!getLocalTime(&timeinfo) && ntpTentatives < 20) {
    delay(500);
    Serial.print(".");
    ntpTentatives++;
  }
  if (getLocalTime(&timeinfo)) {
    Serial.println("\n✓ NTP OK : " + obtenirTimestamp());
  } else {
    Serial.println("\n⚠ NTP non synchronisé — timestamps approximatifs");
  }

  // ── Résumé de la configuration ───────────────────────────────
  Serial.println("===========================================");
  Serial.println("  Station ID    : " + String(STATION_ID));
  Serial.println("  Mesures       : " + CHEMIN_MESURES);
  Serial.println("  Historique    : " + CHEMIN_HISTORIQUE);
  Serial.println("  GPS           : " + CHEMIN_GPS);
  Serial.println("  Intervalle    : " + String(INTERVALLE_MS/1000) + "s");
  Serial.println("===========================================");

  dernierCalcul = millis();
  dernierEnvoi  = millis() - INTERVALLE_MS; // Envoyer dès le démarrage
}


// ============================================================
// LOOP
// ============================================================

void loop() {
  unsigned long maintenant = millis();

  // Reconnexion WiFi automatique si perte de signal
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("⚠ WiFi perdu — reconnexion...");
    WiFi.reconnect();
    delay(3000);
    return;
  }

  if (maintenant - dernierEnvoi >= INTERVALLE_MS) {
    dernierEnvoi = maintenant;

    Serial.println("\n── Lecture capteurs ──────────────────────");

    float temp   = lireTemperature();
    float humAir = lireHumiditeAir();
    float humSol = lireHumiditeSol();
    float vent   = lireVitesseVent();

    Serial.println("═════════════════════════════════════════");
    Serial.println("  Station      : " + String(STATION_ID));
    Serial.printf ("  Température  : %.1f °C\n",    temp);
    Serial.printf ("  Humidité air : %.1f %%\n",    humAir);
    Serial.printf ("  Humidité sol : %.1f %%\n",    humSol);
    Serial.printf ("  Vent         : %.2f km/h\n",  vent);
    Serial.println("═════════════════════════════════════════");

    Serial.println("── Envoi Firebase ────────────────────────");

    // 1 seul PUT pour les mesures actuelles (tout le nœud d'un coup)
    envoyerMesures(temp, humAir, humSol, vent);

    // POST pour l'historique (nouvelle entrée à chaque cycle)
    envoyerHistorique(temp, humAir, humSol, vent);

    // GPS si module présent
#if GPS_ACTIF
    float lat = 0, lon = 0, alt = 0;
    if (lireGPS(lat, lon, alt)) {
      Serial.printf("  GPS : %.6f, %.6f alt=%.1fm\n", lat, lon, alt);
      envoyerGPS(lat, lon, alt);
    } else {
      Serial.println("  GPS : pas de fix");
    }
#endif

    Serial.println("── Cycle terminé ─────────────────────────");
  }
}

#Importe la classe gTTS (Google Text-to-Speech) depuis le module gtts.
#Cette classe permet de convertir du texte en voix.
from gtts import gTTS

#Importe la bibliothèque pygame, souvent utilisée pour créer des jeux, mais ici, 
# on l’utilise pour jouer le fichier audio généré par gTTS.
import pygame

#définissez une variable mytext contenant le texte que vous voulez que l’ordinateur prononce.
#Ce texte est en français.
mytext = 'Bonjour. VIVE Departement MI. je suis entrain de corriger mes papiers des conférences CHIRA 2024 et ICINCO 24!'

#La langue du texte est spécifiée comme français ('fr') pour que la voix synthétique le prononce correctement.
language = 'fr'

#Crée un objet gTTS avec les paramètres :
#text=mytext → le texte à convertir en audio.
#lang=language → la langue à utiliser pour la prononciation.
#slow=False → voix rapide/normale. Si True, la voix serait plus lente.
myobj = gTTS(text=mytext, lang=language, slow=False)

#Convertit le texte en audio et sauvegarde le fichier audio au format MP3 sous le nom "MOI.mp3" 
# dans le répertoire courant.
myobj.save("MOI.mp3")

#Initialise  le système audio de pygame pour pouvoir jouer du son.
pygame.mixer.init()

#Charge le fichier "MOI.mp3" que vous venez de créer afin qu’il soit prêt à être joué.
pygame.mixer.music.load("MOI.mp3")

#Joue le fichier audio : vous entendrez le texte prononcé par une voix synthétique.
pygame.mixer.music.play()
